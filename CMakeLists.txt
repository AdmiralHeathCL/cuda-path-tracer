cmake_minimum_required(VERSION 3.18)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 75 CACHE STRING "CUDA architecture; includes PTX for newer GPUs")
endif()
project(cuda_path_tracer VERSION 1.0 LANGUAGES CXX CUDA)

add_library(ray_tracer_headers INTERFACE)
target_include_directories(ray_tracer_headers INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_include_directories(ray_tracer_headers SYSTEM INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/third_party)
target_compile_features(ray_tracer_headers INTERFACE cxx_std_17)
target_compile_definitions(ray_tracer_headers INTERFACE RAY_TRACER_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
add_library(ray_tracer_dependencies STATIC third_party/stb_image.cpp third_party/tinyobjloader/implementation.cpp)
target_compile_features(ray_tracer_dependencies PRIVATE cxx_std_17)
target_link_libraries(ray_tracer_headers INTERFACE ray_tracer_dependencies)

add_library(ray_tracer_cuda STATIC src/cuda/cuda_renderer.cu)
target_include_directories(ray_tracer_cuda PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
set_target_properties(ray_tracer_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
# Keep host/device arithmetic consistent at grazing intersections and hit ties.
target_compile_options(ray_tracer_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
add_executable(ray_tracer src/app/main.cpp)
target_link_libraries(ray_tracer PRIVATE ray_tracer_headers ray_tracer_cuda)

option(RAY_TRACER_ENABLE_VIEWER "Build the interactive GLFW/OpenGL viewer" ON)
if(RAY_TRACER_ENABLE_VIEWER)
    find_package(glfw3 3.3 REQUIRED)
    find_package(OpenGL REQUIRED)
    add_library(ray_tracer_display INTERFACE)
    target_link_libraries(ray_tracer_display INTERFACE ray_tracer_headers glfw OpenGL::GL)
    target_compile_definitions(ray_tracer_display INTERFACE GL_GLEXT_PROTOTYPES)
    add_library(ray_tracer_cuda_gl STATIC src/cuda/cuda_gl_display.cu)
    set_target_properties(ray_tracer_cuda_gl PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
    target_link_libraries(ray_tracer_cuda_gl PUBLIC ray_tracer_cuda ray_tracer_display)
    add_executable(ray_tracer_viewer src/display/viewer.cpp)
    target_link_libraries(ray_tracer_viewer PRIVATE ray_tracer_cuda_gl)
endif()

include(CTest)
if(BUILD_TESTING)
    function(add_host_test name)
        add_executable(${name} tests/${name}.cpp)
        target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests)
        target_link_libraries(${name} PRIVATE ray_tracer_headers)
        add_test(NAME ${name} COMMAND ${name})
    endfunction()
    foreach(name lighting_tests preview_tests camera_tests triangle_tests mesh_tests obj_tests transport_tests flat_tests)
        add_host_test(${name})
    endforeach()
    add_executable(cuda_tests tests/cuda_tests.cpp)
    target_include_directories(cuda_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests)
    target_link_libraries(cuda_tests PRIVATE ray_tracer_headers ray_tracer_cuda)
    add_test(NAME cuda_tests COMMAND cuda_tests)
    add_executable(cuda_lighting_tests tests/lighting_tests.cpp)
    target_include_directories(cuda_lighting_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests)
    target_link_libraries(cuda_lighting_tests PRIVATE ray_tracer_headers ray_tracer_cuda)
    target_compile_definitions(cuda_lighting_tests PRIVATE RAY_TRACER_HAS_CUDA=1)
    add_test(NAME cuda_lighting_tests COMMAND cuda_lighting_tests)
    set_tests_properties(cuda_tests cuda_lighting_tests PROPERTIES SKIP_RETURN_CODE 77 RUN_SERIAL TRUE)
    add_executable(mesh_validation tests/mesh_validation.cpp)
    target_link_libraries(mesh_validation PRIVATE ray_tracer_headers)
    add_test(NAME mesh_validation COMMAND mesh_validation ${CMAKE_CURRENT_SOURCE_DIR}/assets/models/blender_sphere/sphere.obj 96)
    if(RAY_TRACER_ENABLE_VIEWER)
        foreach(name viewer_input_tests gl_display_tests cuda_gl_tests)
            add_executable(${name} tests/${name}.cpp)
            target_link_libraries(${name} PRIVATE ray_tracer_cuda_gl)
            add_test(NAME ${name} COMMAND ${name})
        endforeach()
        set_tests_properties(gl_display_tests cuda_gl_tests PROPERTIES SKIP_RETURN_CODE 77 RUN_SERIAL TRUE)
    endif()
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME render_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/render_tests.py $<TARGET_FILE:ray_tracer>)
        set_tests_properties(render_tests PROPERTIES SKIP_RETURN_CODE 77 RUN_SERIAL TRUE)
        if(RAY_TRACER_ENABLE_VIEWER)
            add_test(NAME viewer_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/viewer_tests.py $<TARGET_FILE:ray_tracer_viewer> $<TARGET_FILE:ray_tracer>)
            set_tests_properties(viewer_tests PROPERTIES SKIP_RETURN_CODE 77 RUN_SERIAL TRUE)
        endif()
    endif()
endif()
